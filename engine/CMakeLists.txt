cmake_minimum_required(VERSION 3.10)
project(vita_wa_engine C)
set(CMAKE_C_STANDARD 11)
set(MBED_DIR "/tmp/vitaengine/mbedtls-build" CACHE PATH "build di mbedTLS (solo prove su PC)")
set(MBED_SRC "/tmp/vitaengine/mbedtls-3.6.2" CACHE PATH "sorgenti di mbedTLS")
file(STRINGS quickjs/VERSION QJS_VERSION LIMIT_COUNT 1)
add_definitions(-DCONFIG_VERSION="${QJS_VERSION}" -D_GNU_SOURCE)
add_library(qjs STATIC quickjs/quickjs.c quickjs/cutils.c quickjs/dtoa.c quickjs/libregexp.c quickjs/libunicode.c)
target_compile_options(qjs PRIVATE -O2 -w)
target_include_directories(qjs PUBLIC quickjs)
add_executable(engine_pc src/engine.c src/curve25519.c src/image.c src/audio.c src/main_pc.c)
target_include_directories(engine_pc PRIVATE src quickjs ${MBED_SRC}/include)
target_link_libraries(engine_pc qjs ${MBED_DIR}/library/libmbedtls.a ${MBED_DIR}/library/libmbedx509.a ${MBED_DIR}/library/libmbedcrypto.a opus ogg jpeg png webpdemux webp sharpyuv z m pthread)

add_executable(qjsbc tools/qjsbc.c)
target_link_libraries(qjsbc qjs m pthread)
